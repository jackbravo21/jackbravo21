function nome(str) {
    var arr = str.split(' ');
    if (arr[1][0].toUpperCase() != arr[1][0]) arr.splice(1, 1);

- Computed:
* Passando valor/value para a computed property, dentro dela deve haver uma opcao de GET e SET;

<div id="computed">

	<h2>Nome: {{ person.firstName }} {{ person.lastName }}</h2>

	//nova propriedade da instancia view;
	<h2>Nome Completo(computed): {{ fullName }}</h2>


	//dando get no metodo criado fullName;
	<h2>Nome Completo(metodo): {{ getFullName }}</h2>


<script>

	let app = new Vue({
		el: "#computed",
		data: {
			person: {
				firstName: "Jack",
				lastName: "Bravo"
			},
		computed:{
			fullName:{
				get: function() {
					return `${this.person.firstName} ${this.person.lastName}`;
				},
				ser: function(newValue) {
					//split vai pegar o espaco e criar um array entre as duas posicoes
					let name = newValue.split(" ");
					this.person.firstName = name[0];
					this.person.lastName = name[1];
				}
			}
		}
	})

</script>

- Watchers:
* Usado quando temos que escutar alguma propriedade;
* informamos dentro do "watch" o que queremos escutar;

<div id="Watchers">

	//v-model ja eh um watcher por natureza, ele monitora para ver se vai ser modificada a propriedade do data;
	<input type="text" v-model="country>

<script>

	let app = new Vue({
		el: "#watchers",
		data: {
			country: "Brasil"
			},
		watch:{
			//dizemos qual propriedade queremos monitorar, o old mostra qual o valor antigo;
			country: function(newValue, oldValue){
				console.log(newValue, oldValue);
			}
		}
	})

</script>

- Ciclo de vida da instancia Vue;
* Permite executar codigos, antes, durante e depois da aplicacao estar na pagina;

---------------------------------------------

1 - beforeCreate() -> Executada assim que a instância Vue é criada, mas antes de os dados, eventos e watchers estarem disponíveis.
2 - created() -> Executada assim que a instância Vue é criada. Tem à disposição os recursos básicos: dados, computed properties, events, watchers, etc. Contudo, a aplicação ainda não foi montada.
3 - beforeMount() -> Executada um pouco antes do processo de montagem da aplicação começar.
4 - mounted() -> Chamada assim que o processo de montagem é finalizado, já tem acesso à propriedade this.el, para manipular o componente.

1 - activated() -> Chamada quando um componente com a estrutura keep-alive é ativado
2 - deactivated() -> Chamada quando um componente com a estrutura keep-alive é desativado
3 - errorCaptured() -> Chamada quando ocorre um erro em algum componente (Novidade versão 2.5 do Vue)

---------------------------------------------

let app = new Vue({
created: function() {
	console.log(‘A instância foi criada’)
},
mounted: function() {
	console.log(‘A aplicação foi montada’);
}
});

- Trazendo dados ao renderizar:


<div id="lifecycle">

	<h1>Olha a {{ name }}</h1>

	<ul>
		//precisamos criar uma variavel generica para o vue usar para acessar os dados/propriedade do objeto;
		<li v-for="state in states">{{ state }}</li>
	</ul>

<script>

	let app = new Vue({
		el: "#lifecycle",
		data: {
				name: "Informacao na tela",
				states: []
			},
		beforeCreate(){
			console.log("beforeCreate");
		},
		created(){
			//acessa a URL da API do cep do IBGE;
			fetch("https://servicodados.ibge.gov.br/api/v1/localidades/estados")
			.then(response => {
				response.json().then(data =>{
					this.states = data;		//states da pagina vai receber os dados/data;
				})
			}).catch(err => {
				console.log(err);
			})
		},
		beforeMount(){
			console.log("beforeMount");
		},
		mounted(){
			console.log("mounted");
		},
		beforeUpdate(){
			console.log("beforeUpdate");
		},
		updated(){
			console.log("updated");
		},
		beforeDestroy(){
			console.log("beforeDestroy");
		},
		destroyed(){
			console.log("destroyed");
		},
	})

</script>
