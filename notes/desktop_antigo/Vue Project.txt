    <!-- esse titulo vai receber os dados pelo props -->
    <h1>{{titulo}}</h1>    
</div>

</template>

<!-- Depois eu abro e fecho o script -->

<script>
//a primeira coisa que eu faco eh dar um export
export default
{
    name:"teste",
    props:["titulo"]
}

</script>

    color: red;
}
