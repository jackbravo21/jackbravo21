    public Transform frontCheck;                    //circulo invisivel que checa se esta na parede;
    bool wallSliding;                               //verifica se esta na parede ou nao;
    public float wallSlidingSpeed;

- Dentro da funcao Update, vamos declarar a isTouchingFront para nosso circulo que esta em nossa frente, receber a physic que ira checar a mesma coisa que o chao, se esta a frente da parede, o raio, e se esta na parede, detectando a layer Ground;

	isTouchingFront = Physics2D.OverlapCircle(frontCheck.position, checkRadius, whatIsGround);

- Abaixo declaramos uma condicao if para verificar se esta tocando a parede com a frente, se nao esta no chao, e se o input eh diferente de 0, para habilitar ou desabilitar o wallSliding:

        isTouchingFront = Physics2D.OverlapCircle(frontCheck.position, checkRadius, whatIsGround);

        if (isTouchingFront == true && isGrounded == false && input != 0)
        {
            wallSliding = true;
        }
        else
        {
            wallSliding = false;
        }

- Agora sim, abaixo do codigo acima, declaramos o codigo que realmente faz funcionar o wall jump, pegando o rigidbody e adicionando velocidade aos eixos, fazendo com que vc caia mais devagar, dando a sensacao que esta deslizando na parede, sliding on the wall:

        if(wallSliding)
        {
            rb.velocity = new Vector2(rb.velocity.x, Mathf.Clamp(rb.velocity.y, -wallSlidingSpeed, float.MaxValue)); 
        }

- Neste momento devemos ir no player e criar o Objeto vazio "FrontCheck" e colocar na frente dele, e dentro do script Player setar o FrontCheck, com este FrontCheck criado agora, e setar o wallSpeed, aqui setaremos 5;

- Ate agora fizemos apenas para ele deslizar, a partir de agora sera o wallJumping;

- Vamos criar as variaveis necessarias. A wallJumping que verifica se pode ou nao pular na parede (o mesmo que pular no chao), agora duas variaveis publicas para setar a forca nos eixos x e y, e por ultimo o tempo para pular na parede;

    public float xWallForce;
    public float yWallForce;

- Agora abaixo do codigo acima (update), vamos verificar se a tecla esta sendo pressionada no wall, para isso precisa ter a condicao tambem de estar deslizando para poder pular no wall, entao setamos como true, assim como fazemos com o salto normal, mas na parede:

        {
            wallJumping = true;
        }

- No final do codigo vamos criar a funcao SetWallJumpingToFalse():

	{
	    wallJumping = false;
	}

- E agora dentro daquele mesmo if anterior que seta o wallJunping como true, vamos usar a funcao "Invoke()" que recebe 2 argumentos, o 1 eh a Funcao() que queremos executar/rodar em forma de String, e o 2 argumento eh apos quanto tempo eu quero executar essa funcao, passada atraves de uma variavel publica que setara este tempo;

        }

- Continuando abaixo deste if, vamos criar um if que verifica se o wallJumping eh true, e modificar o rigidbody dando velocidade a ele, verificando o x e dando forca negativa, declarando apenas o y;

        }

- Agora basta ir ate o script do player e setar "Wall Jump Time" = "0.05", "X Wall Force" = "15", "Y Wall Force" = "30";
