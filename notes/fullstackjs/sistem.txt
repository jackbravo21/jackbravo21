Passo 1: Configurando o Redux Store
Vamos configurar o store.js com o middleware do Saga.
=============================================================================



/src/redux/store.js
=============================================================================
import { configureStore } from '@reduxjs/toolkit';
import createSagaMiddleware from 'redux-saga';
import userReducer from './user/userSlice'; // Importa o reducer
import userSaga from './user/userSaga'; // Importa a saga

const sagaMiddleware = createSagaMiddleware();

export const store = configureStore({
  reducer: {
    user: userReducer, // Adiciona o reducer
  },
  middleware: (getDefaultMiddleware) => getDefaultMiddleware().concat(sagaMiddleware),
});

sagaMiddleware.run(userSaga);

export default store;

Passo 2: Criando o Slice do Usuário
Crie o userSlice.js para definir o estado do usuário e as ações de login.

/src/redux/user/userSlice.js

=============================================================================
import { createSlice } from '@reduxjs/toolkit';

const userSlice = createSlice({
  name: 'user',
  initialState: {
    isLoggedIn: false,
    userInfo: null,
    error: null,
  },
  reducers: {
    loginRequest: (state) => {
      state.error = null;
    },
    loginSuccess: (state, action) => {
      state.isLoggedIn = true;
      state.userInfo = action.payload;
    },
    loginFailure: (state, action) => {
      state.isLoggedIn = false;
      state.error = action.payload;
    },
  },
});

export const { loginRequest, loginSuccess, loginFailure } = userSlice.actions;
export default userSlice.reducer;

Passo 3: Criando o Saga do Usuário
No userSaga.js, configuramos a função que faz a requisição assíncrona de login.

/src/redux/user/userSaga.js
=============================================================================
import { call, put, takeLatest } from 'redux-saga/effects';
import axios from 'axios';
import { loginRequest, loginSuccess, loginFailure } from './userSlice';

function* handleLogin(action) {
  try {
    const response = yield call(axios.post, 'http://localhost:9000/login', action.payload);
    yield put(loginSuccess(response.data));
  } catch (error) {
    yield put(loginFailure(error.message));
  }
}

function* userSaga() {
  yield takeLatest(loginRequest.type, handleLogin);
}

export default userSaga;

Passo 4: Configurando a Página de Login
Crie uma página de login que dispare a ação de login.

/src/pages/LoginPage.js
=============================================================================
import React, { useState } from 'react';
import { useDispatch, useSelector } from 'react-redux';
import { loginRequest } from '../redux/user/userSlice';

function LoginPage() {
  const [credentials, setCredentials] = useState({ email: '', password: '' });
  const dispatch = useDispatch();
  const error = useSelector((state) => state.user.error);

  const handleChange = (e) => {
    setCredentials({ ...credentials, [e.target.name]: e.target.value });
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    dispatch(loginRequest(credentials));
  };

  return (
    <div>
      <h2>Login</h2>
      <form onSubmit={handleSubmit}>
        <input type="email" name="email" placeholder="Email" onChange={handleChange} />
        <input type="password" name="password" placeholder="Senha" onChange={handleChange} />
        <button type="submit">Entrar</button>
      </form>
      {error && <p>{error}</p>}
    </div>
  );
}

export default LoginPage;

Passo 5: Integrando o store no App

Por fim, integre o store na aplicação.

/src/index.js
=============================================================================
import React from 'react';
import ReactDOM from 'react-dom/client';
import App from './App';
import { Provider } from 'react-redux';
import { store } from './redux/store';

const root = ReactDOM.createRoot(document.getElementById('root'));

root.render(
  <Provider store={store}>
    <App />
  </Provider>
);
