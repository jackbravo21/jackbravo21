- Para definir as rotas do GraphicQL criamos um arquivo no SRC chamado "schema.graphql", ele nao se chama rota, e sim schema;

//tenho 2 tipos de rotas:

type User {			//Crio um tipo para como vai ser o retorno dos usuarios da "type Query" abaixo. (nao eh usuario no banco, e sim o formato do usuario que eh dentro da requisicao dentro da API, o retorno apenas).
	id: ID!			//sempre me refiro ao tipo ID com letra maiuscula.
	name: String!
	email: String!
}

type Query {			//que eh para quando quero buscar uma informacao;
				//o users é do tipo array e tem que ser obrigatorio, colocando o ! atras do array, diz que ele eh obrigado a retornar o array obrigatoriamente, mesmo que seja nulo/vazio.
	users: [Users!]!	//rota users que vai retornar os usuarios, precisamos declarar o tipo de retorno.
	user(id: ID!): User	//posso ter uma query que retorna um unico usuario, que recebe um parametro "id:" que eh do tipo ID! obrigatorio, retornando um unico usuario "User", se nao tiver o id informado no banco, ele nao retorna nada, se for obrigatorio e nao tiver nada, ele retorna vazio.
}

type Mutation {			//que eh para criar, alterar, deletar uma informacao;
				//createUser vai receber um nome que eh uma string obrigatoria, e vai receber um email string obrigatoria, e vai retornar um usuario:User.
	createUser(name: String!, email: String!): User
}

const { GraphQLServer } = require("graphql-yoga");	//troco para const para nao ter que configurar nada de especial.

server.start();

module.exports = {
	Query: {
		users: () => User.find(),				//nao preciso usar async pq o GFQL ja espera promisses.
		user: (_, { id }) => Users.findById(id),		//preciso de 2 parametros, colocarei _ soh para ter um, e no segundo eu recupero o parametro do esquema, e pegar {id}, e passar o id no findbyid(id).
