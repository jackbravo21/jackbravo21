                function Caneta(preco, cor){
                    var color = cor || 'black';
                    this.preco = preco || 1;
                    this.mudarCor = function(cor){
                        if(cor !== 'red' && cor !== 'blue' && cor !== 'black'){
                            return;
                        }
                        color = cor;
                    }

                    this.pegaCor = function(){
                        return color;
                    }
                }

//se nao passar nada, ele ja vai passar as propriedades de criacao;
