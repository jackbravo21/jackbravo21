//instancia a thread e chama o que ela quer executar, no caso a funcao.


        //sao como um semaforo, eles vao liberar ou barrar a thread.
        static ManualResetEvent manual01;
	//o auto reset event, faz automatico essa parada de parar o semaforo ou liberar;
        static AutoResetEvent auto01;        

        static void Main(string[] args)
        {
            //o false representa o vermelho do semaforo, ele vai ficar pausado.
            manual01 = new ManualResetEvent(false);
            auto01 = new AutoResetEvent(false);


            Thread t1 = new Thread(Executa01);
            t1.Start();

            Thread t2 = new Thread(Executa02);
            t2.Start();

            Thread.Sleep(5000);
            //quando eu uso o set, eh como se eu disse pro semaforo que esta verde.
            manual01.Set();
            //para revogar e deixar vermelho o semaforo novamente, eu uso o reset, que vai da esse efeito de espera, das demais threads.
            manual01.Reset();

            auto01.Set();

            Thread.Sleep(5000);
            manual01.Set();
            auto01.Set();

            Console.ReadKey();
	}


//As duas funcoes que vao ser chamadas na thread (executadas);


        static void Executa01()
        {
            Console.WriteLine("01 - Iniciado Executa01.");
            manual01.WaitOne();
            Console.WriteLine("02 - Em execução 01 Executa01.");
            Console.WriteLine("03 - Em execução 02 Executa01.");
            manual01.WaitOne();
            Console.WriteLine("04 - Finalizado Executa01.");
        }
        static void Executa02()
        {
            Console.WriteLine("01 - Iniciado Executa02.");
            auto01.WaitOne();
            Console.WriteLine("02 - Em execução 01 Executa02.");

            auto01.WaitOne();
            Console.WriteLine("03 - Em execução 02 Executa02.");
            Console.WriteLine("04 - Finalizado Executa02.");
        }
